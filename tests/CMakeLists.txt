cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(pj_plugin_smoketest LANGUAGES CXX)
    find_package(plotjuggler_sdk 0.40.0 REQUIRED COMPONENTS plugin_host)
    find_package(Qt6 REQUIRED COMPONENTS Core)
endif()
add_executable(pj_plugin_smoketest pj_plugin_smoketest.cpp)
target_compile_features(pj_plugin_smoketest PRIVATE cxx_std_20)
target_link_libraries(pj_plugin_smoketest PRIVATE plotjuggler_sdk::plugin_host Qt6::Core)
if(TARGET DataAPBin)
    set(_verify --verify)
    if(ADD_UNITS)
        list(APPEND _verify --expect-units)
    endif()
    if(ADD_RCOU_FUNCTION_LABELS)
        list(APPEND _verify --expect-rcou)
    endif()
    add_test(NAME apbin_load_and_lifecycle
        COMMAND pj_plugin_smoketest $<TARGET_FILE:DataAPBin> "${CMAKE_CURRENT_SOURCE_DIR}/logs/00000001.BIN" ${_verify})
    add_test(NAME apbin_metadata_generator
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_metadata_generator.py")
endif()
